function raumbuchPages(intro, cont, outro) {
  const head = '<table class="rbt"><colgroup><col><col style="width:10mm"><col style="width:20mm"><col style="width:11mm"><col style="width:15mm"><col style="width:50mm"><col style="width:16mm"></colgroup><thead><tr><th>Bezeichnung</th><th>Nr</th><th>Bodenbelag</th><th>Dispo</th><th class="num">Fläche m²</th><th>Reinigungsgruppe</th><th class="num">Std./Jahr</th></tr></thead><tbody>';

  rows.push('<tr class="rgt"><td colspan="4">Total</td><td class="num">' + n2(c.area) + '</td><td></td><td class="num">' + n2(c.hours) + '</td></tr>');
  const tblH = rs => head + rs.join('') + '</tbody></table>';
  const pages = []; let cur = [], i = 0, pre = intro;
  while (i < rows.length) {
    cur.push(rows[i]);
    const last = i === rows.length - 1, h = meas(pre + tblH(cur) + (last ? outro : ''));
    if (h > AREA && cur.length > 1) { cur.pop(); while (cur.length > 1 && /class="rg[b1]"/.test(cur[cur.length - 1])) { cur.pop(); i--; } pages.push(pre + tblH(cur)); cur = []; pre = cont; continue; }
    i++;
  }
  pages.push(pre + tblH(cur) + outro);
  return pages;
}

/* ---------- Leistungsverzeichnis: je Reinigungsgruppe ein Block, zweispaltig, läuft über Spalten und Seiten weiter ---------- */
function lvPages(intro, cont) {
  const ch = '<div class="ch"><span>Kategorie, Eintrag</span><span>Code</span><span>Wo</span><span>Mt</span><span>J</span></div>';
  const hgt = h => meas('<div class="alv1">' + h + '</div>', 75);
  const B = D.lv.map(b => { const hd = cn => '<div class="bh">' + e(b.g + '  ' + b.nm) + (cn ? ' <i>Fortsetzung</i>' : '') + '</div>' + ch; const rows = []; let last = '';
    b.ts.forEach(x => { if (x.cat !== last) { rows.push({ html: '<div class="k">' + e(x.cat) + '</div>' }); last = x.cat; } rows.push({ html: '<div class="r"><span><i>✓</i>' + e(x.name).replace(/\//g, '/&#8203;') + (x.nur && x.nur.length ? '<small class="nur">nur: ' + e(x.nur.join(', ')) + '</small>' : '') + '</span><span>' + e(x.code) + '</span><span>' + x.f[0] + '</span><span>' + x.f[1] + '</span><span>' + x.f[2] + '</span></div>' }); });
    if (b.rooms.length > 1) rows.push({ html: '<div class="who">Gilt für: ' + e(b.rooms.join(', ')) + '</div>' });
    rows.forEach(r => r.mm = hgt(r.html)); return { h: hd(false), hc: hd(true), hmm: hgt(hd(false)), rows, tot: hgt(hd(false)) + rows.reduce((a, r) => a + r.mm, 0) }; });
  const GAP = 4, pages = []; let cols = [[]], used = 0, capNow = AREA - meas(intro);
  const newCol = () => { if (cols.length < 2) cols.push([]); else { pages.push(cols); cols = [[]]; capNow = AREA - meas(cont); } used = 0; };
  B.forEach(b => {
    if (used > 0 && used + GAP + b.tot > capNow && b.tot <= capNow) newCol();
    if (used > 0) { cols[cols.length - 1].push('<div class="bsp"></div>'); used += GAP; }
    let k = 0, first = true;
    while (k < b.rows.length) {
      const head = first ? b.h : b.hc, hm = b.hmm;
      if (used + hm + b.rows[k].mm + (k + 1 < b.rows.length ? b.rows[k + 1].mm : 0) > capNow && used > 0) { newCol(); continue; }
      cols[cols.length - 1].push(head); used += hm; first = false;
      while (k < b.rows.length && used + b.rows[k].mm <= capNow) { cols[cols.length - 1].push(b.rows[k].html); used += b.rows[k].mm; k++; }
      if (k < b.rows.length) newCol();
    }
  });
  pages.push(cols);
  return pages.map((cs, i) => (i ? cont : intro) + '<div class="alv">' + cs.map(cc => '<div class="alv1">' + cc.join('') + '</div>').join('') + (cs.length < 2 ? '<div></div>' : '') + '</div>');
}
